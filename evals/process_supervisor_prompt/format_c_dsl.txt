PROCESS coffee_shop_order

# Notation:
#   <id> :: <slug> @ <agent> ON <trigger>[:<tool>]    # → activity definition
#   X --> Y                                          # X is a direct predecessor of Y
#   {X, Y, Z} --> W                                  # AND-join (W needs all of X,Y,Z)
#   X --(label)--> Y                                 # conditional branch
#   *terminal*                                       # marks an end-event activity
#
# An activity fires when an agent emits a message matching its trigger.
# Anything that does not match an activity below is a Violation.

ACTIVITIES
  A01  :: identify_customer_request   @ order_agent            ON message
  A02  :: create_order                @ order_agent            ON tool_call:process_order
  A03  :: check_stock                 @ inventory_agent        ON tool_call:check_inventory
  A04  :: place_food_on_tray          @ inventory_agent        ON tool_call:place_on_tray
  A05  :: brew_coffee                 @ barista_agent          ON tool_call:start_preparation
  A05b :: brew_coffee                 @ barista_agent          ON tool_call:end_preparation
  A06  :: purchase_order              @ order_agent            ON tool_call:calculate_total
  A07  :: handout_order               @ order_agent            ON message    *terminal*
  A08  :: investigate_complaint       @ customer_service_agent ON message
  A09  :: offer_refund                @ customer_service_agent ON tool_call:offer_refund    *terminal*
  A09b :: offer_refund                @ customer_service_agent ON tool_call:offer_partial_refund *terminal*
  A10  :: reject_complaint            @ customer_service_agent ON message    *terminal*

EDGES
  start --> A01
  A01 --(order)--> A02
  A01 --(complaint)--> A08
  A02 --> A03
  A03 --> A04                          # AND-split branch
  A03 --> A05                          # AND-split branch
  A03 --> A06                          # AND-split branch
  A05 --> A05b
  {A04, A05b, A06} --> A07             # AND-join
  A08 --(valid_claim)--> A09
  A08 --(valid_claim)--> A09b
  A08 --(default)--> A10

DISAMBIGUATION  (when message has no tool, multiple activities could match)
  order_agent ON message:
    → A07 if any of {A04, A05, A05b, A06} has already fired in the log
    → A01 otherwise
  customer_service_agent ON message:
    → A10 if A08 has already fired in the log
    → A08 otherwise
  inventory_agent ON message  → no activity exists → Violation
  barista_agent   ON message  → no activity exists → Violation

HANDOFFS  (transfer_to_agent is NOT an activity)
  A handoff terminates the most recent open activity in the source lane.
  Emit: Termination:<that activity's id>:<slug>:via_handoff_to_<target_agent>
  If the source lane has no open activity → Violation.

OUTPUT  (the supervisor MUST emit exactly one line, no prose, no quotes)
  Execution:<id>:<slug>
  Termination:<id>:<slug>:<reason>     # reason ∈ {terminal, via_handoff_to_<agent>}
  Violation:<short_snake_case_reason>
